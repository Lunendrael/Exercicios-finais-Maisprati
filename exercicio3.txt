CONSULTAS

1)
SELECT codigo, titulo, tipo, disponivel
FROM item
ORDER BY titulo;

2)
SELECT u.nome AS usuario, i.titulo AS item
FROM emprestimo e
JOIN usuario u ON u.id = e.usuario_id
JOIN item i ON i.id = e.item_id
WHERE e.data_devolucao IS NULL
ORDER BY u.nome;

3)
SELECT u.nome AS usuario, SUM(e.valor_multa) AS total_multas
FROM usuario u
LEFT JOIN emprestimo e ON e.usuario_id = u.id
GROUP BY u.nome
ORDER BY total_multas DESC;

4)
SELECT i.codigo, i.titulo, i.tipo
FROM item i
LEFT JOIN emprestimo e ON e.item_id = i.id
WHERE e.id IS NULL
ORDER BY i.titulo;
